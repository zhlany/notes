```mysql
DELETE FROM table_name WHERE condition ORDER BY RAND() LIMIT n;
```



## 数据库层

### 创建数据表：

```mysql
Create [temporary] table [if not exists] 表名[create_definition,…] table_options;

Create table 表名(create_definition);

Default default_value     #表示默认值

Auto_increment       #自动增值

Primary key         #设置主键

Reference_definition     #为字段添加备注
```

 

### 记录

```mysql
Create schema 数据库名;       #创建数据库
Create schema if not exists数据库名;    #创建前先判断是否重名
Character set = gbk;         #设置字符集为gbk
Show schema；                #查看所有数据库
Show schema like ‘db_%’;     #查看以db_开头的数据表

Use 数据库名；             #使用数据库
Alter schema 数据库名      #修改数据库
     Default character set gbk        #修改字符集为gbk
     Default collate gbk_chinese_ci;   #修改校对规则
Drop schema 数据库名；    #删除数据库
```



## 表层

### 查看表结构：

```mysql
Show columns from 表名 from 库名；#表结构

Describe（DESC）表名;       #表结构

DESC 表名 列名；          #某列结构
```

 

### 修改表结构：

```mysql
Alter table 表名

add 新字段 数据类型 (null | not null)…, #添加新字段

modify 字段 数据类型 (null | not null)…；#修改字段类型

alter table 表名   #修改字段名

change column 字段 新字段 数据类型 (null | not null)…;

alter table 表名drop 字段；   #删除字段

alter table 表名 rename as 新表名；  #修改表名（**use 库）**

rename table 表名 to 新表名；     #重命名表名（**use 表）**
```

### #复制表

```mysql
Create table 复制表名 like 表名；   #结构一样

Create table 复制表名

As select * from表名；     #结构和数据一起备份
```

### #删除表

```mysql
Drop table 表名；
```

### 表数据的增、删、改

```mysql
#1.插入数据

Insert into 表名 values ();   #插入完整数据

Insert into 表名（字段1,字段2）values()；#部分数据

Insert into 表名（字段1,字段2）#插入多条

 values ( , ), ( , ), ( , );
#1.1数据插入2

insert into 表名 set 字段1='',字段2='',字段3='';

#2.修改数据

Update 表名 Set 字段1,字段2 Where {条件} 

   Order by …   #限定表中的行被修改的次序

   Limit 行数;  #限定修改的行数

 

#3.删除数据

delete 表名 Set 字段1,字段2 Where {条件} 

   Order by …   #限定表中的行被修改的次序

   Limit 行数;   #限定修改的行数

 

Truncate table 表名；  #删除边中所有行（原则：删除后又重新建表）
```
