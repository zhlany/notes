```mysql
# 初始化数据库：
mysqld --initialize --console    -->生成随机密码
ALTER USER 'root'@'localhost' IDENTIFIED BY '新密码';（更改密码）

# 无密码进入mysql   <----新建dos
mysqld --console --skip-grant-tables --shared-memory
 
# 登录mySQL:
mysql -h 主机名 -u 用户名 -p
mysql -u root -p   (登录本机的 MySQL 数据库)

# 修改密码
ALTER USER 'root'@'localhost' IDENTIFIED WITH mysql_native_password BY 'root';
# 启动及关闭 MySQL 服务器:
mysqld --console              启动
mysqladmin -uroot shutdown       关闭
```
