```sh
//永不休眠
vi /etc/profile
TMOUT=0
HISTSIZE=1000                    # 内存中保存的历史记录数量
HISTFILESIZE=2000                # 历史文件保存的记录数量
HISTTIMEFORMAT="%F %T "          # 显示日期和时间 时间戳格式
HISTCONTROL=ignoredups:erasedups # # 忽略重复命令 忽略连续重复，删除所有重复

export GREP_OPTIONS="--color=auto" # grep 添加颜色

source /etc/profile
```
