# 数组、字典

```bash

arr=(10 20 30)
echo "${arr[0]}"       # 输出 20
echo "${arr[@]}"       # 输出 10 20 30
echo "${arr[*]}"       # 等效于 @
echo "${#arr[@]}"      # 输出元素个数

# 数组追加元素
arr+=(40 50)
arr+=("newValue")

# 删除数组内容
unset arr[1]   # 删除索引 1 的元素
unset arr      # 删除整个数组
```

## 1.1遍历数组

```bash
# 普通遍历
for item in "${arr[@]}"; do
    echo "$item"
done

# 使用索引遍历（推荐）
for i in "${!arr[@]}"; do
    echo "index=$i value=${arr[$i]}"
done
```

## 1.2数组切片

```bash
# Bash 支持数组切片：
slice=("${arr[@]:1:2}")   # 从索引 1 开始，取 2 个
echo "${slice[@]}"
```

## 1.3 检查、排序

```bash
# 判断数组是否为空
if [ ${#arr[@]} -eq 0 ]; then
    echo "数组为空"
fi

# Bash 本身不排序，但可借助 sort：
sorted=($(printf "%s\n" "${arr[@]}" | sort))
```



## 2. 字典（关联数组）Bash 4+

```bash
declare -A user

user[name]="Alice"
user[age]=30
user[city]="NewYork"

echo "${user[name]}" # 读取

unset user[age]      # 删除键
unset user           # 删除整个字典

echo "${!user[@]}"   # 查询所有键
echo "${user[@]}"    # 查询所有值
echo "Length: ${#user[@]}"   # 字典长度

# 遍历
for key in "${!user[@]}"; do
    echo "$key : ${user[$key]}"
done

# 判断键是否存在
key="city"
if [[ -v user[$key] ]]; then
    echo "Key exists"
else
    echo "Key not found"
fi

```

## 3. 数组案例

```bash
#!/bin/bash

# 定义
arr=(1 2 3)

# 追加
arr+=(4)

# 修改
arr[1]=20

# 遍历
for i in "${!arr[@]}"; do
    echo "index $i = ${arr[$i]}"
done

# 删除
unset arr[2]

# 长度
echo "Length: ${#arr[@]}"
```

## 4 字典案例

```bash
#!/bin/bash

declare -A person

# 增加
person[name]="Alice"
person[age]=30
person[city]="NY"

# 修改
person[age]=31

# 查询
echo "Name: ${person[name]}"
echo "Age : ${person[age]}"
echo "City: ${person[city]}"

# 遍历
echo "All:"
for key in "${!person[@]}"; do
    echo "$key = ${person[$key]}"
done

# 删除某个键
unset person[city]

# 删除整个字典
# unset person

# 长度
echo "Count: ${#person[@]}"
```



# 返回值

## 1. return 返回状态码（仅限 0–255 的整数）

```bash
myfunc() {
    return 10   # 只能返回整数 0-255
}
# --------------
myfunc
ret=$?
echo "返回值: $ret"
```

## 2. 使用 `echo` 输出值（最常用，用于返回字符串）

```bash
myfunc() {
    echo "hello world"
}
# --------------
value=$(myfunc)
echo "$value"
```

## 3. 使用全局变量或传递变量引用（用于返回多个值）

### 方法 3.1：使用 `echo` 返回多个值

```bash
myfunc() {
    echo "value1 value2"
}

read v1 v2 <<< "$(myfunc)"
echo "$v1 / $v2"

# 或返回多行：
myfunc() {
    echo "line1"
    echo "line2"
}
result=$(myfunc)

```

### 方法 3.2：通过修改全局变量返回多个值

```bash
myfunc() {
    result1="a"
    result2="b"
}
myfunc
echo "$result1 $result2"
```

### 方法 3.3：通过变量名传引用（较专业）

Bash 4+ 支持传变量名，然后使用 `printf -v` 设置值

```bash
myfunc() {
    local _out1=$1
    local _out2=$2

    printf -v "$_out1" "aaa"
    printf -v "$_out2" "bbb"
}

myfunc ret1 ret2
echo "$ret1 | $ret2"
```

## 4. 案例

```bash
案例 1：使用 echo 返回一个值（最常用）
#--------------------------------
#!/bin/bash

calc_sum() {
    local a=$1
    local b=$2
    echo $((a + b))
}

result=$(calc_sum 3 5)
echo "结果：$result"

#--------------------------------
案例 2：返回多个值（通过 echo + read）
#--------------------------------
#!/bin/bash
get_user_info() {
    local name="Alice"
    local age=30
    local city="NewYork"

    echo "$name $age $city"
}
read name age city <<< "$(get_user_info)"
echo "Name: $name"
echo "Age: $age"
echo "City: $city"

#--------------------------------
案例 3：通过“传变量名”方式返回多个值（专业写法，推荐用于大型脚本）
适用于 Bash 4+。
#--------------------------------
#!/bin/bash

get_stats() {
    local _out1=$1
    local _out2=$2
    local _out3=$3

    local cpu="20%"
    local mem="512MB"
    local disk="80%"

    printf -v "$_out1" "$cpu"
    printf -v "$_out2" "$mem"
    printf -v "$_out3" "$disk"
}

get_stats cpu_usage mem_usage disk_usage

echo "CPU: $cpu_usage"
echo "Memory: $mem_usage"
echo "Disk: $disk_usage"
```
